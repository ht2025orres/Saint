                'text-gray-400':            !dia.esMesActual && !dia.esHoy,
                'text-gray-700':             dia.esMesActual && !dia.esHoy
              }">{{ dia.fecha.getDate() }}</span>
          </div>
          <div *ngIf="dia.esMesActual" class="space-y-0.5">
            <div *ngFor="let res of dia.resumenPorUsuario.slice(0, 3)"
              class="flex items-center gap-1 px-1 py-0.5 rounded-md text-xs"
              [ngClass]="getSemaforoResumenClass(res.semaforo)">
              <div class="w-4 h-4 rounded-full flex items-center justify-center text-white font-bold flex-shrink-0"
                style="font-size:9px" [ngClass]="getColorPorId(res.uid)">{{ res.iniciales }}</div>
              <div class="flex items-center gap-1 flex-1 min-w-0">
                <span *ngIf="res.countSeguimiento" class="flex items-center gap-0.5">
                  <i class="bi bi-calendar3" style="font-size:9px"></i><span>{{ res.countSeguimiento }}</span>
                </span>
                <span *ngIf="res.countProyecto" class="flex items-center gap-0.5">
                  <i class="bi bi-kanban" style="font-size:9px"></i><span>{{ res.countProyecto }}</span>
                </span>
                <span *ngIf="res.countGlpi" class="flex items-center gap-0.5">
                  <i class="bi bi-ticket-perforated" style="font-size:9px"></i><span>{{ res.countGlpi }}</span>
                </span>
                <span *ngIf="res.countInforme" class="flex items-center gap-0.5">
                  <i class="bi bi-file-earmark-text" style="font-size:9px"></i><span>{{ res.countInforme }}</span>
                </span>
              </div>
            </div>
            <div *ngIf="dia.resumenPorUsuario.length > 3" class="text-center text-xs text-gray-400 font-medium">
              +{{ dia.resumenPorUsuario.length - 3 }} más
            </div>
          </div>
        </div>
      </div>
    </div>

    <!-- ═══════════════════════════ VISTA SEMANA ═══════════════════════════ -->
    <div *ngIf="vistaModoCalendario === 'semana'"
      class="bg-white rounded-xl border border-gray-200 overflow-hidden">
      <!-- Cabecera días -->
      <div class="grid grid-cols-7 border-b border-gray-100">
        <div *ngFor="let dia of calendarioSemana"
          class="py-2.5 text-center border-r border-gray-100 last:border-r-0 cursor-pointer hover:bg-gray-50 transition"
          (click)="calendarioDiaNav = new Date(dia.fecha); setModoCalendario('dia')">
          <div class="text-xs font-semibold text-gray-400 uppercase">
            {{ diasSemana[(dia.fecha.getDay() + 6) % 7] }}
          </div>
          <div class="text-lg font-bold mt-0.5"
            [ngClass]="dia.esHoy ? 'text-indigo-600' : 'text-gray-800'">
            {{ dia.fecha.getDate() }}
          </div>
          <!-- Badge total tareas -->
          <div *ngIf="dia.tareas.length + dia.tareasExternas.length + dia.tareasInforme.length > 0"
            class="mx-auto mt-1 w-5 h-5 rounded-full bg-indigo-100 text-indigo-700 text-xs font-bold flex items-center justify-center">
            {{ dia.tareas.length + dia.tareasExternas.length + dia.tareasInforme.length }}
          </div>
        </div>
      </div>
      <!-- Contenido por día -->
      <div class="grid grid-cols-7 divide-x divide-gray-100 min-h-[300px]">
        <div *ngFor="let dia of calendarioSemana"
          class="p-2 min-h-[280px] space-y-1 overflow-y-auto max-h-96"
          [ngClass]="dia.esHoy ? 'bg-indigo-50' : 'bg-white'">

          <ng-container *ngIf="dia.esMesActual">
            <!-- Tareas seguimiento -->
            <div *ngFor="let item of dia.tareas.slice(0, 3)"
              class="rounded-lg px-2 py-1 text-xs border cursor-pointer hover:shadow-sm transition"
              [ngClass]="getEstadoBadgeClass(item.tarea.estado)"
              (click)="abrirModalDia(dia, $event); $event.stopPropagation()">
              <div class="flex items-center gap-1 min-w-0">
                <i class="bi bi-calendar3 flex-shrink-0" style="font-size:9px"></i>
                <span class="truncate font-medium"
                  [ngClass]="{'line-through': item.tarea.estado === 'completado'}">
                  {{ item.tarea.titulo }}
                </span>
              </div>
              <div class="text-gray-500 mt-0.5 text-xs">{{ item.nombreUsuario }}</div>
            </div>

            <!-- Tareas externas -->
            <div *ngFor="let t of dia.tareasExternas.slice(0, 2)"
              class="rounded-lg px-2 py-1 text-xs border cursor-pointer hover:shadow-sm transition"
              [ngClass]="t.origen === 'proyecto' ? 'bg-teal-50 text-teal-700 border-teal-200' : 'bg-orange-50 text-orange-700 border-orange-200'"
              (click)="abrirModalDia(dia, $event)">
              <div class="flex items-center gap-1 min-w-0">
                <i [class]="'bi flex-shrink-0 ' + (t.origen === 'proyecto' ? 'bi-kanban' : 'bi-ticket-perforated')" style="font-size:9px"></i>
                <span class="truncate font-medium">{{ t.titulo }}</span>
              </div>
            </div>

            <!-- Tareas informe -->
            <div *ngFor="let t of dia.tareasInforme.slice(0, 1)"
              class="rounded-lg px-2 py-1 text-xs bg-amber-50 text-amber-700 border border-amber-200 cursor-pointer hover:shadow-sm transition"
              (click)="abrirModalDia(dia, $event)">
              <div class="flex items-center gap-1 min-w-0">
                <i class="bi bi-file-earmark-text flex-shrink-0" style="font-size:9px"></i>
                <span class="truncate font-medium">{{ t.titulo }}</span>
              </div>
            </div>

            <!-- Más -->
            <div *ngIf="dia.tareas.length + dia.tareasExternas.length + dia.tareasInforme.length > 5"
              class="text-center text-xs text-gray-400 font-medium cursor-pointer hover:text-indigo-600 transition"
              (click)="abrirModalDia(dia, $event)">
              +{{ (dia.tareas.length + dia.tareasExternas.length + dia.tareasInforme.length) - 5 }} más
            </div>
          </ng-container>
        </div>
      </div>
    </div>

    <!-- ═══════════════════════════ VISTA DÍA ══════════════════════════════ -->
    <div *ngIf="vistaModoCalendario === 'dia'"
      class="bg-white rounded-xl border border-gray-200 overflow-hidden">

      <!-- Stats del día -->
      <div *ngIf="calendarioDiaActual" class="px-5 py-3 border-b border-gray-100 bg-gray-50 flex items-center gap-4 flex-wrap">
        <div *ngFor="let res of (calendarioDiaActual?.resumenPorUsuario ?? [])"
          class="flex items-center gap-2 px-3 py-1.5 rounded-xl border text-sm"
          [ngClass]="getSemaforoResumenClass(res.semaforo)">
          <div class="w-7 h-7 rounded-full flex items-center justify-center text-white font-bold flex-shrink-0"
            [ngClass]="getColorPorId(res.uid)" style="font-size:10px">
            {{ res.iniciales }}
          </div>
          <div>
            <div class="font-semibold text-xs leading-tight">{{ res.nombre }}</div>
            <div class="text-xs">{{ res.completadas }}/{{ res.total }} tareas</div>
          </div>
        </div>
        <div *ngIf="!(calendarioDiaActual?.resumenPorUsuario ?? []).length" class="text-xs text-gray-400">
          Sin tareas para este día
        </div>
      </div>

      <!-- Timeline por horas -->
      <div class="overflow-y-auto max-h-[600px] divide-y divide-gray-50">
        <ng-container *ngFor="let hora of horasDelDia()">
          <div *ngIf="tareasPorHora(calendarioDiaActual, hora).length"
            class="flex gap-3 px-4 py-2.5 hover:bg-gray-50 transition">
            <!-- Hora -->
            <div class="w-12 text-xs text-gray-400 font-mono flex-shrink-0 pt-0.5">
              {{ hora.toString().padStart(2,'0') }}:00
            </div>
            <!-- Tareas en esa hora -->
            <div class="flex-1 space-y-1.5">
              <div *ngFor="let t of tareasPorHora(calendarioDiaActual, hora)"
                class="flex items-center gap-2 px-3 py-1.5 rounded-lg border text-xs"
                [ngClass]="{
                  'bg-indigo-50 border-indigo-200 text-indigo-800': t._tipo === 'seg',
                  'bg-teal-50 border-teal-200 text-teal-800':       t._tipo === 'proyecto',
                  'bg-orange-50 border-orange-200 text-orange-800': t._tipo === 'glpi',
                  'bg-amber-50 border-amber-200 text-amber-800':    t._tipo === 'informe'
                }">
                <i class="flex-shrink-0"
                  [ngClass]="{
                    'bi bi-calendar3':            t._tipo === 'seg',
                    'bi bi-kanban':               t._tipo === 'proyecto',
                    'bi bi-ticket-perforated':    t._tipo === 'glpi',
                    'bi bi-file-earmark-text':    t._tipo === 'informe'
                  }" style="font-size:10px"></i>
                <span class="flex-1 font-medium truncate"
                  [ngClass]="{'line-through opacity-60': t.estado === 'completado'}">
                  {{ t.titulo }}
                </span>
                <span class="px-1.5 py-0.5 rounded-full text-xs font-medium"
                  [ngClass]="getEstadoBadgeClass(t.estado)">{{ t.estado | uppercase }}</span>
                <!-- Avatar usuario -->
                <div *ngIf="t.usuario_id || t._nombre"
                  class="w-5 h-5 rounded-full flex items-center justify-center text-white font-bold flex-shrink-0"
                  [ngClass]="getColorPorId(t.usuario_id ?? 0)"
                  style="font-size:8px"
                  [title]="t._nombre || nombreUsuario(t.usuario_id)">
                  {{ getInicialesResponsable(t.usuario_id ?? 0) }}
                </div>
              </div>
            </div>
          </div>
        </ng-container>

        <!-- Empty state si no hay nada con hora -->
        <div *ngIf="!calendarioDiaActual || (!calendarioDiaActual.tareas.length && !calendarioDiaActual.tareasExternas.length && !calendarioDiaActual.tareasInforme.length)"
          class="text-center py-14">
          <i class="bi bi-calendar-check text-gray-200 text-4xl block mb-3"></i>
          <p class="text-sm text-gray-400">Sin tareas para este día</p>
        </div>

        <!-- Tareas sin hora específica -->
        <div class="px-4 py-3" *ngIf="calendarioDiaActual && (
          calendarioDiaActual.tareas.some(t => !t.tarea.fecha_limite_entrega) ||
          calendarioDiaActual.tareasExternas.some(t => !t.fecha_limite_entrega))">
          <p class="text-xs font-semibold text-gray-400 uppercase tracking-wide mb-2">Sin hora asignada</p>
          <div class="space-y-1.5">
            <div *ngFor="let item of calendarioDiaActual.tareas.filter(t => !t.tarea.fecha_limite_entrega)"
              class="flex items-center gap-2 px-3 py-1.5 rounded-lg border bg-indigo-50 border-indigo-200 text-indigo-800 text-xs">
              <i class="bi bi-calendar3 flex-shrink-0" style="font-size:10px"></i>
              <span class="flex-1 font-medium truncate">{{ item.tarea.titulo }}</span>
            </div>
          </div>
        </div>
      </div>
    </div>

  </ng-container>

</div>
